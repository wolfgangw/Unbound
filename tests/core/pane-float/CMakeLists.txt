cmake_minimum_required(VERSION 3.20)
project(PaneFloatSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(UNBOUND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${CMAKE_CURRENT_SOURCE_DIR}/targets.cmake")
if(WIN32)
    include("${UNBOUND_ROOT}/src/platforms/windows/windowing.cmake")
    unbound_add_pane_tests(UnboundWindowsWindowing "${UNBOUND_ROOT}/tests/platforms/windows/windowing/platformtests.cpp")
    unbound_add_layout_tests(UnboundWindowsWindowing "${UNBOUND_ROOT}/tests/platforms/windows/windowing/layoutadapter.cpp")
    add_executable(audition-activation-test auditionactivationtest.cpp)
    target_link_libraries(audition-activation-test PRIVATE UnboundWindowsWindowing)
elseif(APPLE)
    include("${UNBOUND_ROOT}/src/platforms/macos/windowing.cmake")
    unbound_add_pane_tests(UnboundMacWindowing "${UNBOUND_ROOT}/tests/platforms/macos/platformtests.mm")
    unbound_add_layout_tests(UnboundMacWindowing "${UNBOUND_ROOT}/tests/platforms/macos/layoutadapter.mm")
else()
    message(FATAL_ERROR "Pane fixtures need a native backend and test adapter. See AGENTS.md (platform implementation and validation).")
endif()
