cmake_minimum_required(VERSION 3.20)
project(SoundLibraryFocus LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Widgets)
# Reuse the diagnostic launcher; each command queues work on Resolve's GUI thread.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../inspector-probe/launcher.cpp" launcher)
string(REPLACE "InspectorProbe.dll" "SoundLibraryFocus.dll" launcher "${launcher}")
string(REPLACE "require(std::wstring(command).empty(), L\"Unknown argument. Run Unbound without arguments, or use --version.\");"
    "const ULONG_PTR operation=std::wstring(command)==L\"--passive\" ? 1 : std::wstring(command)==L\"--bridge\" ? 2 : std::wstring(command)==L\"--stop\" ? 3 : 0; require(operation!=0,L\"Use --passive, --bridge or --stop.\");" launcher "${launcher}")
string(REPLACE "nullptr) == 1," "reinterpret_cast<void *>(operation)) == 1," launcher "${launcher}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp" "${launcher}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../audition-events/bridge.h" bridge)
string(REPLACE "unboundFloating_frameAudioPanelMediaPoolContainer" "unboundFloating_frameEditPageSoundLibrary" bridge "${bridge}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/bridge.h" "${bridge}")
add_library(SoundLibraryFocus SHARED probe.cpp)
target_include_directories(SoundLibraryFocus PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(SoundLibraryFocus PRIVATE Qt5::Widgets)
add_executable(sound-library-focus WIN32 "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp")
set_target_properties(sound-library-focus PROPERTIES AUTOMOC OFF MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(sound-library-focus PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNBOUND_VERSION=L"SoundLibraryFocus")
target_link_libraries(sound-library-focus PRIVATE version)
