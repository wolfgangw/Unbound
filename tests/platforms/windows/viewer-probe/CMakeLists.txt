cmake_minimum_required(VERSION 3.20)
project(ViewerProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Viewer probe requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Widgets)
add_library(ViewerProbe SHARED snapshot.cpp)
target_link_libraries(ViewerProbe PRIVATE Qt5::Widgets)
# Reuse the existing diagnostic launcher, keeping a distinct DLL identity so
# inspection can coexist with an already-loaded production helper.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../inspector-probe/launcher.cpp" launcher)
string(REPLACE "InspectorProbe" "ViewerProbe" launcher "${launcher}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp" "${launcher}")
add_executable(viewer-probe WIN32 "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp")
set_property(TARGET viewer-probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(viewer-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNBOUND_VERSION=L"ViewerProbe")
target_link_libraries(viewer-probe PRIVATE version)
