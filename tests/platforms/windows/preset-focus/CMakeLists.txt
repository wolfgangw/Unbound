cmake_minimum_required(VERSION 3.20)
project(PresetFocusProbe LANGUAGES CXX)
set(UNBOUND_FOCUS_MODULE "PresetFocusProbe" CACHE STRING "Unique diagnostic module name")
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Widgets)
add_library(PresetFocusProbe SHARED focustrace.cpp)
set_target_properties(PresetFocusProbe PROPERTIES OUTPUT_NAME "${UNBOUND_FOCUS_MODULE}")
target_link_libraries(PresetFocusProbe PRIVATE Qt5::Widgets)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../inspector-probe/launcher.cpp" launcher)
string(REPLACE "InspectorProbe" "${UNBOUND_FOCUS_MODULE}" launcher "${launcher}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp" "${launcher}")
add_executable(preset-focus-probe WIN32 "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp")
set_property(TARGET preset-focus-probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(preset-focus-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNBOUND_VERSION=L"PresetFocusProbe")
target_link_libraries(preset-focus-probe PRIVATE version)
