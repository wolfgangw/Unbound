cmake_minimum_required(VERSION 3.20)
project(CinemaFocusProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Widgets)
set(PROBE_NAME CinemaFocusProbe CACHE STRING "Unique DLL name for a fresh passive probe")
add_library(${PROBE_NAME} SHARED trace.cpp)
target_link_libraries(${PROBE_NAME} PRIVATE Qt5::Widgets)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../inspector-probe/launcher.cpp" launcher)
string(REPLACE "InspectorProbe" "${PROBE_NAME}" launcher "${launcher}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp" "${launcher}")
add_executable(cinema-focus-probe WIN32 "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp")
set_property(TARGET cinema-focus-probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(cinema-focus-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNBOUND_VERSION=L"CinemaFocusProbe")
target_link_libraries(cinema-focus-probe PRIVATE version)
