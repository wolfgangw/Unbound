cmake_minimum_required(VERSION 3.20)
project(LayoutFocusProbe LANGUAGES CXX)
set(UNBOUND_FOCUS_MODULE "LayoutFocusProbe" CACHE STRING "Unique diagnostic module name")
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Widgets)
add_library(LayoutFocusProbe SHARED focustrace.cpp)
set_target_properties(LayoutFocusProbe PROPERTIES OUTPUT_NAME "${UNBOUND_FOCUS_MODULE}")
target_link_libraries(LayoutFocusProbe PRIVATE Qt5::Widgets)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../inspector-probe/launcher.cpp" launcher)
string(REPLACE "InspectorProbe" "${UNBOUND_FOCUS_MODULE}" launcher "${launcher}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp" "${launcher}")
add_executable(layout-focus-probe WIN32 "${CMAKE_CURRENT_BINARY_DIR}/launcher.cpp")
set_property(TARGET layout-focus-probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(layout-focus-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNBOUND_VERSION=L"LayoutFocusProbe")
target_link_libraries(layout-focus-probe PRIVATE version)
