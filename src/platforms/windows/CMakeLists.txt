if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Unbound requires Windows x64 and MSVC")
endif()
enable_language(RC)
foreach(value UNBOUND_VERSION UNBOUND_BUILD UNBOUND_BUILD_ID)
    if(NOT DEFINED ${value})
        message(FATAL_ERROR "Use infrastructure/windows/build-release.cmd to supply ${value}")
    endif()
endforeach()
include("${UNBOUND_ROOT}/infrastructure/windows/version.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/windowing.cmake")
set(UNBOUND_FILE_TYPE VFT_DLL)
set(UNBOUND_FILENAME Unbound.dll)
configure_file(resources/version.rc.in helper.rc @ONLY)
add_library(UnboundHelper SHARED loading/lua/helperentry.cpp ${CMAKE_CURRENT_BINARY_DIR}/helper.rc)
set_target_properties(UnboundHelper PROPERTIES OUTPUT_NAME Unbound)
set_property(TARGET UnboundHelper PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
unbound_link_core(UnboundHelper UnboundWindowsWindowing)
set(UNBOUND_FILENAME UnboundLuaLoader.dll)
configure_file(resources/version.rc.in loader.rc @ONLY)
add_library(UnboundLuaLoader SHARED loading/lua/loader.cpp ${CMAKE_CURRENT_BINARY_DIR}/loader.rc)
set_property(TARGET UnboundLuaLoader PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(UnboundLuaLoader PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(UnboundLuaLoaderTests "${UNBOUND_ROOT}/tests/platforms/windows/loading/lua/loader.cpp")
target_link_libraries(UnboundLuaLoaderTests PRIVATE Qt5::Widgets)
add_executable(UnboundInstallerTests "${UNBOUND_ROOT}/tests/platforms/windows/installer/main.cpp")
target_include_directories(UnboundInstallerTests PRIVATE installer)
target_compile_definitions(UnboundInstallerTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(UnboundViewerTests "${UNBOUND_ROOT}/tests/platforms/windows/viewer-probe/lifecycle.cpp")
target_link_libraries(UnboundViewerTests PRIVATE UnboundWindowsWindowing)
add_executable(UnboundCinemaTests "${UNBOUND_ROOT}/tests/platforms/windows/cinema-focus/lifecycle.cpp")
set_target_properties(UnboundCinemaTests PROPERTIES AUTOMOC ON)
target_link_libraries(UnboundCinemaTests PRIVATE UnboundWindowsWindowing)
include("${UNBOUND_ROOT}/tests/core/pane-float/targets.cmake")
unbound_add_layout_tests(UnboundWindowsWindowing "${UNBOUND_ROOT}/tests/platforms/windows/windowing/layoutadapter.cpp")
# Preserve the release script's existing Release/ artifact layout.
foreach(target UnboundHelper UnboundLuaLoader UnboundLuaLoaderTests UnboundInstallerTests UnboundViewerTests UnboundCinemaTests UnboundLayoutTests)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
install(TARGETS UnboundLuaLoader UnboundHelper RUNTIME DESTINATION .)
