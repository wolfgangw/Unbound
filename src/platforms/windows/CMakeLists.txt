if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Unbound requires Windows x64 and MSVC")
endif()
enable_language(RC)
foreach(value UNBOUND_VERSION UNBOUND_BUILD UNBOUND_BUILD_ID)
    if(NOT DEFINED ${value})
        message(FATAL_ERROR "Use build-unbound-release.cmd to supply ${value}")
    endif()
endforeach()
include("${UNBOUND_ROOT}/infrastructure/windows/version.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/windowing.cmake")
set(UNBOUND_FILE_TYPE VFT_APP)
set(UNBOUND_FILENAME Unbound.exe)
configure_file(resources/version.rc.in launcher.rc @ONLY)
set(UNBOUND_FILE_TYPE VFT_DLL)
set(UNBOUND_FILENAME Unbound.dll)
configure_file(resources/version.rc.in helper.rc @ONLY)
configure_file(resources/icon.rc.in icon.rc @ONLY)
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/icon.rc PROPERTIES
    OBJECT_DEPENDS "${UNBOUND_ROOT}/src/platforms/windows/resources/unbound.ico")
add_executable(Unbound WIN32 launcher.cpp ${CMAKE_CURRENT_BINARY_DIR}/launcher.rc ${CMAKE_CURRENT_BINARY_DIR}/icon.rc)
set_property(TARGET Unbound PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(Unbound PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNBOUND_VERSION=L"${UNBOUND_VERSION}")
target_link_libraries(Unbound PRIVATE version)
add_executable(UnboundLauncherTests ${UNBOUND_ROOT}/tests/platforms/windows/launcher/main.cpp)
set_property(TARGET UnboundLauncherTests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(UnboundLauncherTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(UnboundLauncherTests PRIVATE version)
add_library(UnboundHelper SHARED "${UNBOUND_ROOT}/src/core/controller.cpp" bootstrap.cpp ${CMAKE_CURRENT_BINARY_DIR}/helper.rc)
set_target_properties(UnboundHelper PROPERTIES OUTPUT_NAME Unbound)
set_property(TARGET UnboundHelper PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(UnboundHelper PRIVATE UnboundWindowsWindowing)
add_executable(UnboundViewerTests "${UNBOUND_ROOT}/tests/platforms/windows/viewer-probe/lifecycle.cpp")
target_link_libraries(UnboundViewerTests PRIVATE UnboundWindowsWindowing)
add_executable(UnboundPresetTests "${UNBOUND_ROOT}/tests/core/pane-float/presettest.cpp" "${UNBOUND_ROOT}/src/core/controller.cpp")
target_link_libraries(UnboundPresetTests PRIVATE UnboundWindowsWindowing)
# Preserve the release script's existing Release/ artifact layout.
foreach(target Unbound UnboundHelper UnboundLauncherTests UnboundViewerTests UnboundPresetTests)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
install(TARGETS Unbound UnboundHelper RUNTIME DESTINATION .)
