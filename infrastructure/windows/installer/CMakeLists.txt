cmake_minimum_required(VERSION 3.20)
project(UnboundInstaller LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(UNBOUND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${UNBOUND_ROOT}/src/installer/dialog.json" "${UNBOUND_ROOT}/infrastructure/installer/generate.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${UNBOUND_ROOT}/infrastructure/installer/generate.py"
    "${UNBOUND_ROOT}/src/installer/dialog.json" "${CMAKE_CURRENT_BINARY_DIR}/installer-ui.generated.h"
    COMMAND_ERROR_IS_FATAL ANY)
include("${UNBOUND_ROOT}/infrastructure/windows/version.cmake")
set(UNBOUND_FILE_TYPE VFT_APP)
set(UNBOUND_INSTALLER_NAME "Unbound ${UNBOUND_MAJOR}.${UNBOUND_MINOR}.${UNBOUND_PATCH}")
if(UNBOUND_VERSION MATCHES "\\+internal\\.([0-9][0-9][0-9][0-9])([0-9][0-9])([0-9][0-9])\\.([0-9]+)$")
    set(_date "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
    set(_counter "${CMAKE_MATCH_4}")
    string(LENGTH "${_counter}" _length)
    while(_length LESS 3)
        set(_counter "0${_counter}")
        string(LENGTH "${_counter}" _length)
    endwhile()
    string(APPEND UNBOUND_INSTALLER_NAME " (${_date}.${_counter})")
endif()
set(UNBOUND_FILENAME "${UNBOUND_INSTALLER_NAME}.exe")
configure_file("${UNBOUND_ROOT}/src/platforms/windows/resources/version.rc.in" version.rc @ONLY)
configure_file("${UNBOUND_ROOT}/src/platforms/windows/resources/icon.rc.in" icon.rc @ONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/payload-version.txt" "${UNBOUND_VERSION}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/payload.rc"
    "201 RCDATA \"${UNBOUND_PAYLOAD}/UnboundLuaLoader.dll\"\n202 RCDATA \"${UNBOUND_PAYLOAD}/Unbound.dll\"\n203 RCDATA \"${UNBOUND_ROOT}/src/platforms/windows/loading/lua/startup/Unbound.scriptlib\"\n204 RCDATA \"${CMAKE_CURRENT_BINARY_DIR}/payload-version.txt\"\n")
add_executable(UnboundInstaller WIN32 "${UNBOUND_ROOT}/src/platforms/windows/installer/main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/version.rc" "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
target_include_directories(UnboundInstaller PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/payload.rc" "205 RCDATA \"${UNBOUND_ROOT}/LICENSE\"\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/installer-name.txt" "${UNBOUND_INSTALLER_NAME}")
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/payload.rc" "206 RCDATA \"${CMAKE_CURRENT_BINARY_DIR}/installer-name.txt\"\n")
set_property(TARGET UnboundInstaller PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(UnboundInstaller PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(UnboundInstaller PRIVATE comctl32)
add_executable(UnboundInstallerStateTests "${UNBOUND_ROOT}/tests/platforms/windows/installer/statetest.cpp")
target_include_directories(UnboundInstallerStateTests PRIVATE "${UNBOUND_ROOT}/src/platforms/windows/installer")
target_include_directories(UnboundInstallerStateTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(UnboundInstallerStateTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(UnboundInstallerStateTests PRIVATE comctl32)
set_property(TARGET UnboundInstallerStateTests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(UnboundInstallTool install.cpp)
add_executable(UnboundInstallerTests "${UNBOUND_ROOT}/tests/platforms/windows/installer/main.cpp")
target_include_directories(UnboundInstallerTests PRIVATE "${UNBOUND_ROOT}/src/platforms/windows/installer")
target_compile_definitions(UnboundInstallerTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET UnboundInstallerTests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_include_directories(UnboundInstallTool PRIVATE "${UNBOUND_ROOT}/src/platforms/windows/installer")
target_compile_definitions(UnboundInstallTool PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET UnboundInstallTool PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_link_options(UnboundInstaller PRIVATE "/MANIFESTDEPENDENCY:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='amd64' publicKeyToken='6595b64144ccf1df' language='*'")
